\section{Antiderivación de una Función Compuesta}

Sea $g$ una función cuyo recorrido o rango es un intervalo $I$, y sea $f$ una función continua en $I$. Si $g$ es derivable en su dominio y $F$ es una antiderivada o primitiva de $f$ en $I$, entonces:

Si $u = g(x)$, entonces $du = g'(x)\,dx$, por lo que:
\begin{equation*}
    \int f(g(x))g'(x)\,dx = F(g(x)) + C
\end{equation*}
\begin{equation*}
    \int f(u)\,du = F(u) + C
\end{equation*}

\begin{tcolorbox}
    El método de sustitución utiliza la \textbf{regla de la cadena a la inversa}. 
    Esta técnica de integración se aplica cuando el integrando está compuesto por dos expresiones donde una de ellas es la derivada de la otra (pudiendo diferir únicamente por un factor constante).
\end{tcolorbox}

\subsection*{Ejemplo comparativo: Relación entre Integral y Derivada}

\begin{center}
\begin{tabular}{|c|c|}
    \hline
    \textbf{Integral} & \textbf{Derivada} \\ \hline
    $\displaystyle \int 3t^2 \cos(t^3)\,dt$ & $\displaystyle \frac{d}{dt}\left[\sin(t^3)\right]$ \\ \hline
\end{tabular}
\end{center}

Calculando la derivada:
\begin{equation*}
    \frac{d}{dt}\left[\sin(t^3)\right] = 3t^2 \cos(t^3)
\end{equation*}

Puesto que el argumento $\sin(t^3)$ tiene como derivada a $3t^2 \cos(t^3)$, concluimos que:
\begin{equation*}
    \int \underbrace{3t^2}_{\substack{\text{Derivada de la}\\\text{función interna}}} \cos(\underbrace{t^3}_{\substack{\text{Función}\\\text{interna}}})\,dt = \sin(t^3) + C
\end{equation*}

\textbf{Observación:} Notar que la función compuesta en el integrando cuenta con una función exterior $f(u) = \cos(u)$ y una función interior $g(t) = t^3$. La derivada $g'(t) = 3t^2$ figura directamente como factor del integrando.

\section{Selección y Análisis de Integrales por Sustitución Directa}

\textbf{Ejercicio 2:} Seleccionar únicamente las integrales que cumplan las siguientes dos condiciones:
\begin{enumerate}[label=\arabic*.]
    \item Están compuestas por dos expresiones.
    \item Una expresión es la derivada de la otra (pudiendo diferir únicamente por un valor constante).
\end{enumerate}

\begin{multicols}{2}
\begin{enumerate}[label=\alph*.]
    \item $\displaystyle \int 3x^2\sqrt{x^3+1}\,dx$
    \item $\displaystyle \int (2x+3)(x^2+x)\,dx$
    \item $\displaystyle \int \sin^2(3x)\cos(3x)\,dx$
    \item $\displaystyle \int x\sqrt{x-1}\,dx$
    \item $\displaystyle \int \frac{\sqrt{9-x^2}}{x^2}\,dx$
    \item $\displaystyle \int x^3(x^2+1)^4\,dx$
    \item $\displaystyle \int 2x e^x\,dx$
    \item $\displaystyle \int \frac{-4x}{(1-2x^2)^2}\,dx$
    \item $\displaystyle \int x\sin(x)\,dx$
    \item $\displaystyle \int \frac{x+5}{x^2+x-2}\,dx$
\end{enumerate}
\end{multicols}

\subsection*{Análisis de selección}
\begin{itemize}
    \item[\textbf{a.}] $\displaystyle \int 3x^2\sqrt{x^3+1}\,dx \implies u = x^3+1 \implies du = 3x^2\,dx$ \quad \textbf{(Cumple)}
    \item[\textbf{b.}] $\displaystyle \int (2x+3)(x^2+x)\,dx \implies u = x^2+x \implies du = (2x+1)\,dx \neq (2x+3)\,dx$ \quad \textbf{(No cumple)}
    \item[\textbf{c.}] $\displaystyle \int \sin^2(3x)\cos(3x)\,dx \implies u = \sin(3x) \implies du = 3\cos(3x)\,dx \iff \frac{du}{3} = \cos(3x)\,dx$ \quad \textbf{(Cumple)}
    \item[\textbf{d.}] $\displaystyle \int x\sqrt{x-1}\,dx \implies u = x-1 \implies du = dx, \; x = u+1$ \quad \textbf{(Requiere despeje algebraico)}
    \item[\textbf{e.}] $\displaystyle \int \frac{\sqrt{9-x^2}}{x^2}\,dx \implies u = 9-x^2 \implies du = -2x\,dx$ \quad \textbf{(No cumple, requiere sustitución trigonométrica)}
    \item[\textbf{f.}] $\displaystyle \int x^3(x^2+1)^4\,dx \implies u = x^2+1 \implies \frac{du}{2} = x\,dx, \; x^2 = u-1$ \quad \textbf{(Requiere despeje algebraico)}
    \item[\textbf{g.}] $\displaystyle \int 2x e^x\,dx \implies$ Requiere integración por partes \quad \textbf{(No cumple)}
    \item[\textbf{h.}] $\displaystyle \int \frac{-4x}{(1-2x^2)^2}\,dx \implies u = 1-2x^2 \implies du = -4x\,dx$ \quad \textbf{(Cumple)}
    \item[\textbf{i.}] $\displaystyle \int x\sin(x)\,dx \implies$ Requiere integración por partes \quad \textbf{(No cumple)}
    \item[\textbf{j.}] $\displaystyle \int \frac{x+5}{x^2+x-2}\,dx \implies u = x^2+x-2 \implies du = (2x+1)\,dx$ \quad \textbf{(No cumple directamente, fracciones parciales)}
\end{itemize}

\textbf{Observación:} En los casos directos analizados, la expresión seleccionada como $u$ corresponde a la parte interna que contiene el mayor grado o exponente respecto a su diferencial.

\section{Solución Detallada de las Integrales}

\subsection*{Caso a:}
\begin{equation*}
    \int 3x^2\sqrt{x^3+1}\,dx
\end{equation*}
Tomamos:
\begin{align*}
    u &= x^3 + 1 \\
    du &= 3x^2\,dx
\end{align*}
Sustituyendo en la integral:
\begin{align*}
    \int \sqrt{u}\,du &= \int u^{1/2}\,du \\
    &= \frac{2}{3}u^{3/2} + C \\
    &= \frac{2\sqrt{u^3}}{3} + C \\
    &= \frac{2\sqrt{(x^3+1)^3}}{3} + C
\end{align*}

\subsection*{Caso c:}
\begin{equation*}
    \int \sin^2(3x)\cos(3x)\,dx
\end{equation*}
Tomamos:
\begin{align*}
    u &= \sin(3x) \\
    du &= 3\cos(3x)\,dx \implies \frac{du}{3} = \cos(3x)\,dx
\end{align*}
Sustituyendo:
\begin{align*}
    \int u^2 \left(\frac{du}{3}\right) &= \frac{1}{3}\int u^2\,du \\
    &= \frac{1}{3}\left(\frac{u^3}{3}\right) + C \\
    &= \frac{u^3}{9} + C \\
    &= \frac{\sin^3(3x)}{9} + C
\end{align*}

\subsection*{Caso d (Sustitución con despeje algebraico):}
\begin{equation*}
    \int x\sqrt{x-1}\,dx
\end{equation*}
Tomamos:
\begin{align*}
    u &= x - 1 \implies x = u + 1 \\
    du &= dx
\end{align*}
Sustituyendo:
\begin{align*}
    \int (u+1)\sqrt{u}\,du &= \int (u^{3/2} + u^{1/2})\,du \\
    &= \frac{2}{5}u^{5/2} + \frac{2}{3}u^{3/2} + C \\
    &= \frac{2\sqrt{u^5}}{5} + \frac{2\sqrt{u^3}}{3} + C \\
    &= \frac{2\sqrt{(x-1)^5}}{5} + \frac{2\sqrt{(x-1)^3}}{3} + C
\end{align*}

\subsection*{Caso f (Sustitución con partición y despeje):}
\begin{equation*}
    \int x^3(x^2+1)^4\,dx = \int x^2(x^2+1)^4(x\,dx)
\end{equation*}
Tomamos:
\begin{align*}
    u &= x^2 + 1 \implies x^2 = u - 1 \\
    du &= 2x\,dx \implies x\,dx = \frac{du}{2}
\end{align*}
Sustituyendo:
\begin{align*}
    \int (u-1)u^4 \left(\frac{du}{2}\right) &= \frac{1}{2}\int (u^5 - u^4)\,du \\
    &= \frac{1}{2}\left(\frac{u^6}{6} - \frac{u^5}{5}\right) + C \\
    &= \frac{u^6}{12} - \frac{u^5}{10} + C \\
    &= \frac{(x^2+1)^6}{12} - \frac{(x^2+1)^5}{10} + C
\end{align*}

\subsection*{Caso h:}
\begin{equation*}
    \int \frac{-4x}{(1-2x^2)^2}\,dx
\end{equation*}
Tomamos:
\begin{align*}
    u &= 1 - 2x^2 \\
    du &= -4x\,dx
\end{align*}
Sustituyendo:
\begin{align*}
    \int \frac{du}{u^2} &= \int u^{-2}\,du \\
    &= -u^{-1} + C \\
    &= -\frac{1}{u} + C \\
    &= -\frac{1}{1-2x^2} + C
\end{align*}

\section{Algoritmo para Realizar un Cambio de Variable}

\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{Elegir una sustitución $u = g(x)$:} Usualmente es conveniente seleccionar la parte interna de una función compuesta:
    \begin{itemize}
        \item Una cantidad elevada a una potencia o exponente.
        \item El argumento de una función trigonométrica.
        \item El argumento de una función logarítmica o exponencial.
        \item La cantidad subradical (radicando).
    \end{itemize}
    \item \textbf{Calcular el diferencial:} Hallar $du = g'(x)\,dx$.
    \item \textbf{Reescribir la integral:} Expresar completamente el integrando en términos de la variable $u$ y su diferencial $du$.
    \item \textbf{Integrar:} Resolver la integral elemental en términos de $u$.
    \item \textbf{Retornar a la variable original:} Reemplazar $u$ por $g(x)$ para obtener la antiderivada en términos de $x$.
    \item \textbf{Comprobar:} Verificar el resultado derivando la antiderivada obtenida.
\end{enumerate}
