\section{Fundamentos y Razones Trigonométricas}

El objetivo principal de las sustituciones trigonométricas es eliminar el radical presente en el integrando mediante el uso de identidades pitagóricas:
\begin{align*}
    \sin^2(x) + \cos^2(x) &= 1 \\
    \sec^2(x) &= \tan^2(x) + 1 \\
    \tan^2(x) &= \sec^2(x) - 1
\end{align*}

\subsection*{Razones trigonométricas en el triángulo rectángulo}
Dado un triángulo rectángulo con cateto opuesto ($co$), cateto adyacente ($ca$) e hipotenusa ($h$):
\begin{multicols}{3}
\begin{itemize}
    \item $\displaystyle \sin(\alpha) = \frac{co}{h}$
    \item $\displaystyle \cos(\alpha) = \frac{ca}{h}$
    \item $\displaystyle \tan(\alpha) = \frac{co}{ca}$
    \item $\displaystyle \csc(\alpha) = \frac{h}{co}$
    \item $\displaystyle \sec(\alpha) = \frac{h}{ca}$
    \item $\displaystyle \cot(\alpha) = \frac{ca}{co}$
\end{itemize}
\end{multicols}

Por el \textbf{Teorema de Pitágoras}:
\begin{equation*}
    p^2 + l^2 = h^2
\end{equation*}

\subsection*{Casos estándar de sustitución trigonométrica}

\begin{center}
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{|c|c|c|c|}
    \hline
    \textbf{Expresión en el integrando} & \textbf{Sustitución} & \textbf{Diferencial ($dx$)} & \textbf{Identidad utilizada} \\ \hline
    $\sqrt{a^2 - x^2}$ & $x = a\sin(\theta)$ & $dx = a\cos(\theta)\,d\theta$ & $a^2 - a^2\sin^2(\theta) = a^2\cos^2(\theta)$ \\ \hline
    $\sqrt{a^2 + x^2}$ & $x = a\tan(\theta)$ & $dx = a\sec^2(\theta)\,d\theta$ & $a^2 + a^2\tan^2(\theta) = a^2\sec^2(\theta)$ \\ \hline
    $\sqrt{x^2 - a^2}$ & $x = a\sec(\theta)$ & $dx = a\sec(\theta)\tan(\theta)\,d\theta$ & $a^2\sec^2(\theta) - a^2 = a^2\tan^2(\theta)$ \\ \hline
\end{tabular}
\end{center}

\section{Ejemplos Desarrollados}

\subsection*{Ejemplo 1: Caso $\sqrt{a^2 - x^2}$}
Evaluar la integral indefinida:
\begin{equation*}
    \int \frac{dx}{x^2\sqrt{9 - x^2}}
\end{equation*}

\textbf{Definición del triángulo rectángulo:}
\begin{itemize}
    \item Hipotenusa: $h = 3$
    \item Cateto opuesto: $co = x$
    \item Cateto adyacente: $ca = \sqrt{9 - x^2}$
    \item $\displaystyle \sin(\theta) = \frac{x}{3} \implies x = 3\sin(\theta)$
    \item $\displaystyle dx = 3\cos(\theta)\,d\theta$
\end{itemize}

\textbf{Sustitución y simplificación en términos de $\theta$:}
\begin{align*}
    \int \frac{dx}{x^2\sqrt{9 - x^2}} &= \int \frac{3\cos(\theta)\,d\theta}{(3\sin(\theta))^2 \sqrt{9 - (3\sin(\theta))^2}} \\
    &= \int \frac{3\cos(\theta)\,d\theta}{9\sin^2(\theta) \sqrt{9 - 9\sin^2(\theta)}} \\
    &= \int \frac{3\cos(\theta)\,d\theta}{9\sin^2(\theta) \sqrt{9[1 - \sin^2(\theta)]}} \\
    &= \int \frac{3\cos(\theta)\,d\theta}{9\sin^2(\theta) \cdot 3\sqrt{1 - \sin^2(\theta)}} \\
    &= \int \frac{3\cos(\theta)\,d\theta}{27\sin^2(\theta)\sqrt{\cos^2(\theta)}} \\
    &= \int \frac{3\cos(\theta)\,d\theta}{27\sin^2(\theta)\cos(\theta)} \\
    &= \frac{1}{9}\int \frac{d\theta}{\sin^2(\theta)} \\
    &= \frac{1}{9}\int \csc^2(\theta)\,d\theta \\
    &= -\frac{1}{9}\cot(\theta) + C
\end{align*}

\textbf{Retorno a la variable original $x$:}
A partir del triángulo de referencia, $\cot(\theta) = \frac{ca}{co} = \frac{\sqrt{9-x^2}}{x}$:
\begin{equation*}
    \int \frac{dx}{x^2\sqrt{9 - x^2}} = -\frac{1}{9}\left(\frac{\sqrt{9 - x^2}}{x}\right) + C = -\frac{\sqrt{9 - x^2}}{9x} + C
\end{equation*}

\subsection*{Ejemplo 2: Caso $\sqrt{a^2 + x^2}$}
Evaluar la integral indefinida:
\begin{equation*}
    \int \frac{dx}{\sqrt{4x^2 + 1}}
\end{equation*}

\textbf{Definición del triángulo rectángulo:}
\begin{itemize}
    \item Cateto opuesto: $co = 2x$
    \item Cateto adyacente: $ca = 1$
    \item Hipotenusa: $h = \sqrt{4x^2 + 1}$
    \item $\displaystyle \tan(\theta) = \frac{2x}{1} = 2x \implies x = \frac{\tan(\theta)}{2}$
    \item $\displaystyle dx = \frac{\sec^2(\theta)}{2}\,d\theta$
\end{itemize}

\textbf{Sustitución y simplificación en términos de $\theta$:}
\begin{align*}
    \int \frac{dx}{\sqrt{4x^2 + 1}} &= \int \frac{\frac{\sec^2(\theta)}{2}}{\sqrt{4\left(\frac{\tan(\theta)}{2}\right)^2 + 1}}\,d\theta \\
    &= \int \frac{\frac{\sec^2(\theta)}{2}}{\sqrt{4\left(\frac{\tan^2(\theta)}{4}\right) + 1}}\,d\theta \\
    &= \int \frac{\frac{\sec^2(\theta)}{2}}{\sqrt{\tan^2(\theta) + 1}}\,d\theta \\
    &= \int \frac{\frac{\sec^2(\theta)}{2}}{\sqrt{\sec^2(\theta)}}\,d\theta \\
    &= \int \frac{\frac{\sec^2(\theta)}{2}}{\sec(\theta)}\,d\theta \\
    &= \frac{1}{2}\int \sec(\theta)\,d\theta \\
    &= \frac{1}{2}\ln|\sec(\theta) + \tan(\theta)| + C
\end{align*}

\textbf{Retorno a la variable original $x$:}
Del triángulo de referencia, $\sec(\theta) = \frac{h}{ca} = \frac{\sqrt{4x^2+1}}{1} = \sqrt{4x^2+1}$ y $\tan(\theta) = 2x$:
\begin{equation*}
    \int \frac{dx}{\sqrt{4x^2 + 1}} = \frac{1}{2}\ln\left|\sqrt{4x^2 + 1} + 2x\right| + C
\end{equation*}

\subsection*{Ejemplo 3: Caso $\sqrt{x^2 - a^2}$}
Evaluar la integral indefinida:
\begin{equation*}
    \int \frac{dx}{(x^2 - 1)^{3/2}} = \int \frac{dx}{\sqrt{(x^2 - 1)^3}}
\end{equation*}

\textbf{Definición del triángulo rectángulo:}
\begin{itemize}
    \item Hipotenusa: $h = x$
    \item Cateto adyacente: $ca = 1$
    \item Cateto opuesto: $co = \sqrt{x^2 - 1}$
    \item $\displaystyle \sec(\theta) = \frac{x}{1} = x$
    \item $\displaystyle dx = \sec(\theta)\tan(\theta)\,d\theta$
\end{itemize}

\textbf{Sustitución y simplificación en términos de $\theta$:}
\begin{align*}
    \int \frac{dx}{\sqrt{(x^2 - 1)^3}} &= \int \frac{\sec(\theta)\tan(\theta)}{\sqrt{(\sec^2(\theta) - 1)^3}}\,d\theta \\
    &= \int \frac{\sec(\theta)\tan(\theta)}{\sqrt{(\tan^2(\theta))^3}}\,d\theta \\
    &= \int \frac{\sec(\theta)\tan(\theta)}{\sqrt{(\tan^3(\theta))^2}}\,d\theta \\
    &= \int \frac{\sec(\theta)\tan(\theta)}{\tan^3(\theta)}\,d\theta \\
    &= \int \frac{\sec(\theta)}{\tan^2(\theta)}\,d\theta \\
    &= \int \frac{\frac{1}{\cos(\theta)}}{\frac{\sin^2(\theta)}{\cos^2(\theta)}}\,d\theta \\
    &= \int \frac{\cos^2(\theta)}{\cos(\theta)\sin^2(\theta)}\,d\theta \\
    &= \int \frac{\cos(\theta)}{\sin^2(\theta)}\,d\theta
\end{align*}

Aplicando sustitución simple:
\begin{align*}
    u &= \sin(\theta) \implies du = \cos(\theta)\,d\theta \\
    \int \frac{\cos(\theta)}{\sin^2(\theta)}\,d\theta &= \int \frac{du}{u^2} = \int u^{-2}\,du = -u^{-1} + C = -\frac{1}{\sin(\theta)} + C
\end{align*}

\textbf{Retorno a la variable original $x$:}
Dado que $\sin(\theta) = \frac{co}{h} = \frac{\sqrt{x^2 - 1}}{x}$:
\begin{equation*}
    \int \frac{dx}{(x^2 - 1)^{3/2}} = -\frac{1}{\frac{\sqrt{x^2 - 1}}{x}} + C = -\frac{x}{\sqrt{x^2 - 1}} + C
\end{equation*}

\section{Formulario de Derivadas e Integrales Trigonométricas}

\subsection*{Derivadas de funciones trigonométricas}
\begin{multicols}{2}
\begin{itemize}
    \item $\displaystyle \frac{d}{dx}[\sin(x)] = \cos(x)$
    \item $\displaystyle \frac{d}{dx}[\cos(x)] = -\sin(x)$
    \item $\displaystyle \frac{d}{dx}[\tan(x)] = \sec^2(x)$
    \item $\displaystyle \frac{d}{dx}[\cot(x)] = -\csc^2(x)$
    \item $\displaystyle \frac{d}{dx}[\sec(x)] = \sec(x)\tan(x)$
    \item $\displaystyle \frac{d}{dx}[\csc(x)] = -\csc(x)\cot(x)$
\end{itemize}
\end{multicols}

\subsection*{Integrales básicas e inmediatas}
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|l|l|}
    \hline
    $\displaystyle \int \sin(x)\,dx = -\cos(x) + C$ & $\displaystyle \int \cos(x)\,dx = \sin(x) + C$ \\ \hline
    $\displaystyle \int \tan(x)\,dx = \ln|\sec(x)| + C$ & $\displaystyle \int \cot(x)\,dx = \ln|\sin(x)| + C$ \\ \hline
    $\displaystyle \int \sec(x)\,dx = \ln|\sec(x) + \tan(x)| + C$ & $\displaystyle \int \csc(x)\,dx = \ln|\csc(x) - \cot(x)| + C$ \\ \hline
    $\displaystyle \int \sec^2(x)\,dx = \tan(x) + C$ & $\displaystyle \int \csc^2(x)\,dx = -\cot(x) + C$ \\ \hline
    $\displaystyle \int \sec(x)\tan(x)\,dx = \sec(x) + C$ & $\displaystyle \int \csc(x)\cot(x)\,dx = -\csc(x) + C$ \\ \hline
\end{tabular}
\end{center}

\section{Ejercicios Propuestos}

Evaluar las siguientes integrales indefinidas y verificar la solución derivando:

\begin{enumerate}[label=\alph*.]
    \item $\displaystyle \int \csc(x)\,dx$
    \item $\displaystyle \int \frac{1}{x\sqrt{4x^2 + 9}}\,dx$
    \item $\displaystyle \int \frac{\sqrt{x^2 - 3}}{x}\,dx$
    \item $\displaystyle \int -\frac{x}{\sqrt{x^2 - 1}}\,dx$
    \item $\displaystyle \int \frac{\sqrt{x^2 - 2x - 3}}{x - 1}\,dx$
    \item $\displaystyle \int \sqrt{9 - x^2}\,dx$
    \item $\displaystyle \int x\sqrt{9 - x^2}\,dx$
    \item $\displaystyle \int \frac{x}{\sqrt{x^2 - 4}}\,dx$
    \item $\displaystyle \int \frac{x^4}{\sqrt{1 - x^2}}\,dx$
\end{enumerate}

---

\section{Soluciones de los Ejercicios Propuestos}

\begin{enumerate}[label=\alph*.]
    \item $\displaystyle \ln|\csc(x) - \cot(x)| + C$
    \item $\displaystyle \frac{1}{3}\ln\left|\frac{\sqrt{4x^2 + 9}}{2x} - \frac{3}{2x}\right| + C$
    \item $\displaystyle \sqrt{3}\left(\frac{\sqrt{x^2 - 3}}{\sqrt{3}} - \sec^{-1}\left(\frac{x}{\sqrt{3}}\right)\right) + C$
    \item $\displaystyle -\sqrt{x^2 - 1} + C$
    \item $\displaystyle \sqrt{(x - 1)^2 - 4} - 2\sec^{-1}\left(\frac{x - 1}{2}\right) + C$
    \item $\displaystyle \frac{9}{2}\sin^{-1}\left(\frac{x}{3}\right) + \frac{x\sqrt{9 - x^2}}{2} + C$
    \item $\displaystyle -\frac{\sqrt{(9 - x^2)^3}}{3} + C$
    \item $\displaystyle \sqrt{x^2 - 4} + C$
    \item $\displaystyle \frac{3}{8}\sin^{-1}(x) - \frac{3x\sqrt{1 - x^2}}{8} - \frac{x^3\sqrt{1 - x^2}}{4} + C$
\end{enumerate}
