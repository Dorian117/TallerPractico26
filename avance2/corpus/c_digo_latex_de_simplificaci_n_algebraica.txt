\subsection*{Desarrollo Paso a Paso}

\subsubsection*{Ejercicio a)}
\[
\int \frac{1+2t-t^{2}}{\sqrt{t}}\,dt
\]

\begin{enumerate}
    \item Expresar la función racional como una suma/resta de funciones racionales (término a término entre $\sqrt{t}$):
    \[
    = \int \left( \frac{1}{\sqrt{t}} + \frac{2t}{\sqrt{t}} - \frac{t^{2}}{\sqrt{t}} \right) dt
    \]
    \item Separar la integral en una combinación lineal de integrales:
    \[
    = \int \frac{1}{\sqrt{t}}\,dt + 2\int \frac{t}{\sqrt{t}}\,dt - \int \frac{t^{2}}{\sqrt{t}}\,dt
    \]
    \item Aplicar propiedades de la potenciación para reescribir los integrandos con exponentes racionales:
    \[
    = \int t^{-1/2}\,dt + 2\int t\cdot t^{-1/2}\,dt - \int t^{2}\cdot t^{-1/2}\,dt
    \]
    \item Simplificar los productos de potencias de igual base e integrar el primer término:
    \[
    = 2t^{1/2} + C_1 + 2\int t^{1/2}\,dt - \int t^{3/2}\,dt
    \]
    \item Integrar los términos restantes aplicando la regla de la potencia:
    \begin{align*}
    &= 2\sqrt{t} + C_1 + 2\left(\frac{2}{3}t^{3/2}\right) + D - \frac{2}{5}t^{5/2} + E \\
    &= 2\sqrt{t} + \frac{4}{3}\sqrt{t^{3}} - \frac{2}{5}\sqrt{t^{5}} + F
    \end{align*}
    \item Expresando bajo un denominador común o simplificando raíces:\footnote{Si se escala por el factor común de denominadores o se extraen factores fuera del radical:}
    \begin{align*}
    &= \frac{30\sqrt{t} + 20\sqrt{t^{3}} - 6\sqrt{t^{5}}}{15} + F \\
    &= \frac{30\sqrt{t} + 20t\sqrt{t} - 6t^{2}\sqrt{t}}{15} + F
    \end{align*}
\end{enumerate}

\subsubsection*{Ejercicio b)}
\[
\int (x^{3}+5)(x^{3}-5)\,dx
\]
\begin{enumerate}
    \item Desarrollar el producto notable (diferencia de cuadrados):
    \[
    = \int (x^{6} - 25)\,dx
    \]
    \item Expresar como la resta de dos integrales inmediatas:
    \[
    = \int x^{6}\,dx - 25\int dx
    \]
    \item Integrar término a término:
    \[
    = \frac{x^{7}}{7} - 25x + C
    \]
\end{enumerate}

\subsubsection*{Ejercicio c)}
\[
\int \frac{x^{4}}{\sqrt{x}}\,dx
\]
\begin{enumerate}
    \item Reescribir el denominador en forma de exponente fraccionario:
    \[
    = \int \frac{x^{4}}{x^{1/2}}\,dx
    \]
    \item Aplicar propiedades de potencias de igual base ($x^{a}/x^{b} = x^{a-b}$):
    \[
    = \int x^{4}\cdot x^{-1/2}\,dx = \int x^{7/2}\,dx
    \]
    \item Integrar mediante la regla de la potencia:
    \[
    = \frac{x^{7/2 + 1}}{\frac{7}{2} + 1} + C = \frac{2x^{9/2}}{9} + C = \frac{2\sqrt{x^{9}}}{9} + C = \frac{2x^{4}\sqrt{x}}{9} + C
    \]
\end{enumerate}

\subsubsection*{Ejercicio d)}
\[
\int \sqrt[3]{x}\left(x^{2} - \frac{4}{3}x\right) dx
\]
\begin{enumerate}
    \item Convertir el radical a exponente fraccionario:
    \[
    = \int x^{1/3}\left(x^{2} - \frac{4}{3}x\right) dx
    \]
    \item Aplicar la propiedad distributiva:
    \[
    = \int x^{7/3}\,dx - \frac{4}{3}\int x^{4/3}\,dx
    \]
    \item Integrar término a término:
    \begin{align*}
    &= \frac{3x^{10/3}}{10} - \frac{4}{3}\left(\frac{3x^{7/3}}{7}\right) + C \\
    &= \frac{3\sqrt[3]{x^{10}}}{10} - \frac{4x^{7/3}}{7} + C \\
    &= \frac{3\sqrt[3]{x^{10}}}{10} - \frac{4\sqrt[3]{x^{7}}}{7} + C \\
    &= \frac{3x^{3}\sqrt[3]{x}}{10} - \frac{4x^{2}\sqrt[3]{x}}{7} + C
    \end{align*}
\end{enumerate}

\newpage

\subsubsection*{Ejercicio e)}
\[
\int \frac{\cos(x)}{\sin^{2}(x)}\,dx
\]
\begin{enumerate}
    \item Descomponer el integrando en factores convenientes:
    \[
    = \int \frac{\cos(x)}{\sin(x)}\cdot\frac{1}{\sin(x)}\,dx
    \]
    \item Utilizar identidades trigonométricas básicas:
    \[
    \frac{\cos(x)}{\sin(x)} = \cot(x), \quad \frac{1}{\sin(x)} = \csc(x)
    \]
    \item Reescribir la integral:
    \[
    = \int \cot(x)\csc(x)\,dx
    \]
    \item Como $\frac{d}{dx}[\csc(x)] = -\csc(x)\cot(x)$, se tiene:
    \[
    = -\csc(x) + C
    \]
\end{enumerate}

\subsubsection*{Ejercicio f)}
\[
\int \frac{dx}{1+\sin(x)}
\]
\begin{enumerate}
    \item Multiplicar numerador y denominador por el conjugado del denominador ($1-\sin(x)$):
    \[
    = \int \left(\frac{1}{1+\sin(x)}\right)\left(\frac{1-\sin(x)}{1-\sin(x)}\right) dx = \int \frac{1-\sin(x)}{1-\sin^{2}(x)}\,dx
    \]
    \item Emplear la identidad pitagórica $\sin^{2}(x) + \cos^{2}(x) = 1 \implies 1-\sin^{2}(x) = \cos^{2}(x)$:
    \[
    = \int \frac{1-\sin(x)}{\cos^{2}(x)}\,dx
    \]
    \item Separar la fracción en dos términos:
    \[
    = \int \frac{1}{\cos^{2}(x)}\,dx - \int \frac{\sin(x)}{\cos^{2}(x)}\,dx
    \]
    \item Expresar en términos de secante y tangente:
    \begin{align*}
    &= \int \sec^{2}(x)\,dx - \int \frac{\sin(x)}{\cos(x)}\cdot\frac{1}{\cos(x)}\,dx \\
    &= \int \sec^{2}(x)\,dx - \int \tan(x)\sec(x)\,dx
    \end{align*}
    \item Integrar directamente recordando que $(\tan(x))' = \sec^{2}(x)$ y $(\sec(x))' = \sec(x)\tan(x)$:
    \[
    = \tan(x) - \sec(x) + C
    \]
\end{enumerate}

\node[largebox, below=1.0cm of rewrite] (methods) {
\textbf{Estrategias para reescribir:}
\begin{itemize}[leftmargin=15pt, itemsep=3pt]
    \item Reescribir una función racional como suma y/o resta de funciones racionales.
    \item Reescribir una función racional como un producto de funciones racionales.
    \item Reescribir un cociente de potencias de igual base como producto de potencias de igual base.
    \item Desarrollar los factores (productos notables, distributiva).
    \item Usar propiedades de la potenciación y de la radicación.
    \item Multiplicar por el conjugado.
    \item Factorizar numerador y denominador de una fracción.
    \item Aplicar identidades trigonométricas.
\end{itemize}
};
